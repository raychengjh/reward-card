      function todayISO() {
        const d = new Date();
        const pad = (n) => String(n).padStart(2, '0');

      function encodeStateToHash(obj) {
        try {
          const json = JSON.stringify(obj);
          const bytes = new TextEncoder().encode(json);
          let bin = '';
          bytes.forEach((b) => { bin += String.fromCharCode(b); });
          return '#card=' + encodeURIComponent(btoa(bin));
        } catch (e) {
          return '';
        }
      }

      function decodeStateFromHash(hashStr) {
        if (!hashStr || !hashStr.startsWith('#card=')) return null;
        try {
          const b64 = decodeURIComponent(hashStr.slice(6));
          const bin = atob(b64);
          const bytes = Uint8Array.from(bin, (c) => c.charCodeAt(0));
          const json = new TextDecoder().decode(bytes);
          const parsed = JSON.parse(json);
          if (parsed && typeof parsed === 'object') {
            return normalizeState(parsed);
          }
        } catch (e) {
          return null;
        }
        return null;
      }

      function normalizeState(raw) {
        return {

        };
      }

        const end = rawText.lastIndexOf('}');

        }
        return JSON.parse(rawText.slice(start, end + 1));

      function escapeHtml(str) {
        return String(str)
          .replace(/&/g, '&amp;')
          .replace(/</g, '&lt;')
          .replace(/>/g, '&gt;')
          .replace(/"/g, '&quot;');

      function handleStampClick(idx) {
        const key = String(idx);
        const existing = state.stamps[key];

        if (!existing) {
          if (state.readOnly) return;
          const prevCount = getStampCount();
          state.stamps[key] = { date: todayISO(), by: '', note: '' };
          lastStampedIndex = idx;
          persistState();
          const nextCount = getStampCount();
          renderAll();
          checkMilestoneCelebration(prevCount, nextCount);
        } else {
          if (state.readOnly) {
            const details = [
              `Stamp #${idx}`,
              existing.date ? `on ${existing.date}` : '',
              existing.by ? `by ${existing.by}` : '',
              existing.note ? `— ${existing.note}` : ''
            ].filter(Boolean).join(' ');
            showToast(details);
            return;
          }
          openStampModal(idx);
        }
      }

          if (n === 10) return `<span class="ml-1.5 text-[11px] font-bold px-2 py-0.5 rounded-full bg-[#fdf1eb] text-[#b44b32]">Small Present 🎁</span>`;

          if (n === 50) return `<span class="ml-1.5 text-[11px] font-bold px-2 py-0.5 rounded-full bg-[#e6f6ee] text-[#236347]">Giant Present 🏆</span>`;
          return '';
        };

        container.innerHTML = `
          <div class="divide-y divide-[#f2e8df]">
            ${entries.map((item) => `
              <div class="flex flex-wrap items-center justify-between gap-3 px-4 py-3 hover:bg-[#fffaf5] transition">
                <div class="flex items-center gap-3 min-w-0">
                  <div class="w-9 h-9 rounded-xl bg-[#fdf1eb] text-[#c85a3f] font-extrabold text-xs flex items-center justify-center shrink-0">#${item.num}</div>
                  <div class="min-w-0">
                    <div class="text-sm font-semibold text-[#2c221e] flex flex-wrap items-center">
                      <span>${item.by ? escapeHtml(item.by) : 'Stamp Collected'}</span>
                      ${milestoneLabel(item.num)}
                    </div>
                    <div class="text-xs text-[#76655d] truncate">
                      ${item.note ? escapeHtml(item.note) : 'No note added'}
                    </div>
                  </div>
                </div>
                <div class="flex items-center gap-2 shrink-0">
                  <span class="text-xs font-medium text-[#8c786e]">${escapeHtml(item.date || '')}</span>
                  ${!state.readOnly ? `<button type="button" data-edit-log="${item.num}" class="no-print text-xs font-semibold text-[#c85a3f] hover:underline px-2 py-1">Edit</button>` : ''}
                </div>
              </div>
            `).join('')}
          </div>
        `;

        container.querySelectorAll('[data-edit-log]').forEach((btn) => {
          btn.addEventListener('click', () => {
            openStampModal(Number(btn.getAttribute('data-edit-log')));
          });
        });
      }

        editingStampIndex = idx;

        document.getElementById('modalStampNum').textContent = `#${idx}`;
        document.getElementById('modalStampTitle').textContent = `Stamp #${idx} Details`;
        document.getElementById('stampByInput').value = entry.by || '';
        document.getElementById('stampNoteInput').value = entry.note || '';
        document.getElementById('stampDateInput').value = entry.date || todayISO();

        modal.classList.remove('hidden');
        modal.classList.add('flex');

        editingStampIndex = null;

      function updateShareLinkField() {
        const readOnlyShare = document.getElementById('shareReadOnlyCheckbox').checked;
        const payload = { ...state, readOnly: readOnlyShare };
        const hash = encodeStateToHash(payload);
        const base = window.location.href.split('#')[0];
        document.getElementById('shareLinkInput').value = base + hash;
      }

      function copyTextSafe(text) {
        if (navigator.clipboard && navigator.clipboard.writeText) {
          navigator.clipboard.writeText(text).then(
            () => showToast('Share link copied to clipboard!'),
            () => fallbackCopy(text)
          );
        } else {
          fallbackCopy(text);
        }
      }

      function fallbackCopy(text) {
        const ta = document.createElement('textarea');
        ta.value = text;
        ta.style.position = 'fixed';
        ta.style.opacity = '0';
        document.body.appendChild(ta);
        ta.select();
        try {
          document.execCommand('copy');
          showToast('Share link copied to clipboard!');
        } catch (e) {
          showToast('Select the link above to copy it.');
        }
        document.body.removeChild(ta);
      }
